import { createClient } from "npm:@supabase/supabase-js@2.95.0"; const FLUTTERWAVE_API_URL = "https://api.flutterwave.com/v3/payments"; const MAX_AMOUNT_NGN = 10_000_000; const corsHeaders = { "Access-Control-Allow-Origin": "", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", "Content-Type": "application/json", }; Deno.serve(async (req: Request) => { const allowedOrigins = (Deno.env.get("ALLOWED_ORIGINS") ?? "") .split(",") .map((value) => value.trim()) .filter(Boolean); const origin = req.headers.get("Origin") ?? ""; if (req.method === "OPTIONS") { return new Response("ok", { headers: { ...corsHeaders, "Access-Control-Allow-Origin": allowedOrigins.includes(origin) ? origin : allowedOrigins[0] ?? "null", }, }); } const headers = { ...corsHeaders, "Access-Control-Allow-Origin": allowedOrigins.includes(origin) ? origin : allowedOrigins[0] ?? "null", }; const respond = (status: number, body: object) => new Response(JSON.stringify(body), { status, headers }); let paymentId: string | undefined; try { if (req.method !== "POST") { return respond(405, { error: "Method not allowed." }); } if (!allowedOrigins.includes(origin)) { return respond(403, { error: "Origin not allowed." }); } const supabaseUrl = Deno.env.get("SUPABASE_URL"); const anonKey = Deno.env.get("SUPABASE_ANON_KEY"); const serviceRoleKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); const flutterwaveKey = Deno.env.get("FLUTTERWAVE_SECRET_KEY"); const redirectUrl = Deno.env.get("FLUTTERWAVE_REDIRECT_URL"); if ( !supabaseUrl || !anonKey || !serviceRoleKey || !flutterwaveKey || !redirectUrl ) { return respond(500, { error: "Server configuration is incomplete." }); } if (!flutterwaveKey.startsWith("FLWSECK_TEST-")) { return respond(500, { error: "This function is configured for Flutterwave test mode.", }); } const authHeader = req.headers.get("Authorization"); if (!authHeader?.startsWith("Bearer ")) { return respond(401, { error: "Please sign in first." }); } const userClient = createClient(supabaseUrl, anonKey, { global: { headers: { Authorization: authHeader } }, auth: { persistSession: false, autoRefreshToken: false }, }); const token = authHeader.slice("Bearer ".length); const { data: { user }, error: authError } = await userClient.auth.getUser(token); if (authError || !user || !user.email) { return respond(401, { error: "Your session is invalid. Please sign in again." }); } const body = await req.json(); const amount = Number(body.amount); if ( !Number.isFinite(amount) || amount < 1 || amount > MAX_AMOUNT_NGN || Math.round(amount * 100) !== amount * 100 ) { return respond(400, { error: "Enter a valid amount in naira." }); } const safeRedirect = new URL(redirectUrl); if ( safeRedirect.protocol !== "https:" || safeRedirect.origin !== "https://ajoks-payhub.pages.dev" ) { return respond(500, { error: "Redirect URL configuration is invalid." }); } const admin = createClient(supabaseUrl, serviceRoleKey, { auth: { persistSession: false, autoRefreshToken: false }, }); const txRef = `ajoks_${crypto.randomUUID()}`; const { data: payment, error: insertError } = await admin .from("payments") .insert({ user_id: user.id, provider: "flutterwave", tx_ref: txRef, amount, currency: "NGN", status: "pending", }) .select("id") .single(); if (insertError || !payment) { console.error("Pending payment record creation failed:", insertError); return respond(500, { error: "Could not create the payment record." }); } paymentId = payment.id; const flutterwaveResponse = await fetch(FLUTTERWAVE_API_URL, { method: "POST", headers: { Authorization: `Bearer ${flutterwaveKey}`, "Content-Type": "application/json", }, body: JSON.stringify({ tx_ref: txRef, amount, currency: "NGN", redirect_url: safeRedirect.toString(), customer: { email: user.email, }, customizations: { title: "Ajoks PayHub", description: "Wallet funding", }, }), }); const result = await flutterwaveResponse.json(); if ( !flutterwaveResponse.ok || result.status !== "success" || typeof result.data?.link !== "string" ) { await admin.from("payments") .update({ status: "failed" }) .eq("id", payment.id); console.error("Flutterwave checkout initialization failed."); return respond(502, { error: "Could not start checkout. Please try again." }); } return respond(200, { checkout_url: result.data.link, tx_ref: txRef, }); } catch (error) { console.error("Payment initialization failed:", error); return respond(500, { error: "An unexpected error occurred while starting payment.", }); } });
